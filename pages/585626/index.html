def _get_minio_client_and_endpoint() -> tuple[Minio, str]:
    endpoint = MinioConfig.minio_endpoint.strip()
    if not endpoint:
        raise ServiceException(message='MINIO_ENDPOINT未配置')

    normalized_endpoint = endpoint
    if normalized_endpoint.startswith('http://'):
        normalized_endpoint = normalized_endpoint[len('http://') :]
    elif normalized_endpoint.startswith('https://'):
        normalized_endpoint = normalized_endpoint[len('https://') :]

    if not MinioConfig.minio_access_key or not MinioConfig.minio_secret_key:
        raise ServiceException(message='MINIO访问密钥未配置')

    client = Minio(
        endpoint=normalized_endpoint,
        access_key=MinioConfig.minio_access_key,
        secret_key=MinioConfig.minio_secret_key,
        secure=MinioConfig.minio_secure,
    )

    return client, normalized_endpoint


@certification_controller.post(
    '/upload',
    summary='上传证书文件接口',
    description='用于上传证书文件到MinIO并返回文件访问地址',
    response_model=DynamicResponseModel[UploadResponseModel],
    dependencies=[UserInterfaceAuthDependency('certification:certification:add')],
)
async def upload_certification_file(request: Request, file: Annotated[UploadFile, File(...)]) -> Response:
    if not file.filename:
        raise ServiceException(message='文件名不能为空')

    bucket_name = MinioConfig.minio_bucket_name.strip()
    if not bucket_name:
        raise ServiceException(message='MINIO_BUCKET_NAME未配置')

    minio_client, endpoint = _get_minio_client_and_endpoint()

    now = datetime.now()
    file_ext = file.filename.rsplit('.', 1)[-1] if '.' in file.filename else ''
    object_name = f'certification/{now.strftime("%Y/%m/%d")}/{uuid4().hex}'
    if file_ext:
        object_name = f'{object_name}.{file_ext}'

    file_content = await file.read()
    if not file_content:
        raise ServiceException(message='上传文件不能为空')

    try:
        if not minio_client.bucket_exists(bucket_name):
            minio_client.make_bucket(bucket_name)

        minio_client.put_object(
            bucket_name=bucket_name,
            object_name=object_name,
            data=BytesIO(file_content),
            length=len(file_content),
            content_type=file.content_type or 'application/octet-stream',
        )
    except S3Error as e:
        logger.error(f'MinIO上传失败: {e}')
        raise ServiceException(message='文件上传失败') from e

    protocol = 'https' if MinioConfig.minio_secure else 'http'
    file_url = f'{protocol}://{endpoint}/{bucket_name}/{object_name}'
    response_data = UploadResponseModel(
        file_name=object_name,
        new_file_name=object_name.rsplit('/', 1)[-1],
        original_filename=file.filename,
        url=file_url,
    )
    logger.info('证书文件上传成功')

    return ResponseUtil.success(model_content=response_data)

const toUploadFileList = (url?: string): UploadFile[] => {
    if (!url) {
        return [];
    }

    const fileName = decodeURIComponent(url.split('/').pop() || 'certificate-file');
    return [
        {
            uid: url,
            name: fileName,
            status: 'done',
            url,
            response: { data: { url } },
        },
    ];
};

const extractUploadedUrl = (value?: UploadFile[] | string): string | undefined => {
    if (!value) {
        return undefined;
    }

    if (typeof value === 'string') {
        return value;
    }

    const firstFile = value[0];
    if (!firstFile) {
        return undefined;
    }

    const response = firstFile.response as UploadApiResponse | undefined;
    return response?.data?.url || response?.url || firstFile.url;
};

      <ProTable<CertificationItem, SearchFormValues>
        rowKey="id"
        actionRef={actionRef}
        columns={columns}
        pagination={{
          pageSize: 10,
          showSizeChanger: true,
        }}
        rowSelection={{
          selectedRowKeys,
          onChange: (keys) => setSelectedRowKeys(keys),
        }}
        request={async (params) => {
          const payload: QueryParams = {
            pageNum: params.current || 1,
            pageSize: params.pageSize || 10,
            certificationNo: params.certificationNo,
            reportNo: params.reportNo,
            productName: params.productName,
            productSerial: params.productSerial,
            factory: params.factory,
            issuingAuthority: params.issuingAuthority,
            fileUrl: params.fileUrl,
            startDate: toDateString(params.startDate),
            expiredTime: toDateString(params.expiredTime),
            status: params.status,
            deptId: params.deptId,
          };

          setLastQuery(payload);
          const response = (await listCertification(payload)) as ListResponse;

          return {
            data: response.rows || [],
            success: true,
            total: response.total || 0,
          };
        }}
        toolBarRender={() => {
          const actions: React.ReactNode[] = [];

          if (hasPermission('certification:certification:add')) {
            actions.push(
              <Button key="add" type="primary" icon={<PlusOutlined />} onClick={handleAdd}>
                新增
              </Button>,
            );
          }

          if (hasPermission('certification:certification:edit')) {
            actions.push(
              <Button key="edit" icon={<EditOutlined />} disabled={single} onClick={() => void handleUpdate()}>
                修改
              </Button>,
            );
          }

          if (hasPermission('certification:certification:remove')) {
            actions.push(
              <Button key="delete" danger icon={<DeleteOutlined />} disabled={multiple} onClick={() => handleDelete()}>
                删除
              </Button>,
            );
          }

          if (hasPermission('certification:certification:export')) {
            actions.push(
              <Button key="export" icon={<DownloadOutlined />} onClick={() => void handleExport()}>
                导出
              </Button>,
            );
          }

          return actions;
        }}
        search={{
          labelWidth: 'auto',
          searchText: '搜索',
          resetText: '重置',
        }}
        options={{
          reload: true,
          density: false,
          setting: false,
          search: false,
        }}
        dateFormatter={false}
      />

      <ModalForm<CertificationFormValues>
        title={title}
        open={open}
        initialValues={formInitialValues}
        onOpenChange={setOpen}
        modalProps={{
          destroyOnClose: true,
          onCancel: () => setOpen(false),
        }}
        onFinish={async (values) => {
          const fileUrl = extractUploadedUrl(values.fileUrl);
          const payload: CertificationItem = {
            ...values,
            fileUrl,
            startDate: toDateString(values.startDate as Dayjs | string | undefined),
            expiredTime: toDateString(values.expiredTime as Dayjs | string | undefined),
          };

          if (payload.id !== undefined) {
            await updateCertification(payload);
            message.success('修改成功');
          } else {
            await addCertification(payload);
            message.success('新增成功');
          }

          setOpen(false);
          await actionRef.current?.reload();
          return true;
        }}
        submitter={{
          searchConfig: {
            submitText: '确定',
            resetText: '取消',
          },
          resetButtonProps: {
            onClick: () => setOpen(false),
          },
        }}
      >
        <ProFormText name="id" hidden />
        <ProFormText
          name="certificationNo"
          label="证书编号"
          placeholder="请输入证书编号"
          rules={[{ required: true, message: '证书编号不能为空' }]}
        />
        <ProFormText name="reportNo" label="报告编号" placeholder="请输入报告编号" />
        <ProFormText name="productName" label="产品名称" placeholder="请输入产品名称" />
        <ProFormText name="productSerial" label="产品型号" placeholder="请输入产品型号" />
        <ProFormText name="factory" label="制造单位" placeholder="请输入制造单位" />
        <ProFormText name="issuingAuthority" label="发证机构" placeholder="请输入发证机构" />
        <ProFormUploadButton
          name="fileUrl"
          label="证书文件"
          max={1}
          fieldProps={uploadFieldProps}
          extra="上传后将自动填充可访问的文件链接"
        />
        <ProFormDatePicker
          name="startDate"
          label="证件生效日期"
          fieldProps={{ format: 'YYYY-MM-DD' }}
          rules={[{ required: true, message: '证件生效日期不能为空' }]}
        />
        <ProFormDatePicker
          name="expiredTime"
          label="证件过期日期"
          fieldProps={{ format: 'YYYY-MM-DD' }}
        />
        <ProFormRadio.Group name="status" label="状态" options={statusOptions} />
        <ProFormText
          name="deptId"
          label="部门ID"
          placeholder="请输入部门ID"
          rules={[{ required: true, message: '部门ID不能为空' }]}
        />
      </ModalForm>
    </PageContainer>
  );
};

export default CertificationPage;
