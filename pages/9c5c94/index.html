# 定义实现方法

@classmethod
    async def batch_import_control_cabinet_info_services(
        cls,
        query_db: AsyncSession,
        file: UploadFile,
    ) -> CrudResponseModel:
        """
        批量导入控制柜数据信息service

        :param query_db: orm对象
        :param file: 上传文件对象
        :return: 导入结果
        """
        header_dict = {
            '合同号': 'contract_no',
            '合同编号': 'contract_no',
            'control_cabinet_contract_no': 'contract_no',
            '控制柜型号': 'control_cabinet_model',
            '控制柜编号': 'control_cabinet_no',
            '控制柜生产厂家': 'control_cabinet_factory',
            '控制柜制造单位': 'control_cabinet_factory',
            '控制柜型式试验证书': 'control_cabinet_certification_no',
            '控制柜型式证书编号': 'control_cabinet_certification_no',
            '控制柜生产日期': 'control_cabinet_create_date',
            '数据范围': 'data_scope',
            '菜单树选择项是否关联显示': 'menu_check_strictly',
            '部门树选择项是否关联显示': 'dept_check_strictly',
            '角色状态': 'status',
            '状态': 'status',
            '备注': 'remark',
        }

        df = await ExcelImportCommon.read_excel_by_header(
            file=file,
            header_dict=header_dict,
            required_fields=['contract_no'],
            missing_header_map={'contract_no': '合同号'},
        )

        try:
            for excel_row_no, (_index, row) in enumerate(df.iterrows(), start=2):
                try:
                    contract_no = ExcelImportCommon.normalize_text(row.get('contract_no'))
                    if not contract_no:
                        raise ServiceException(message='合同号不能为空')

                    add_control_cabinet_info = Control_cabinet_infoModel(
                        contract_no=contract_no,
                        control_cabinet_model=ExcelImportCommon.normalize_text(row.get('control_cabinet_model')),
                        control_cabinet_no=ExcelImportCommon.normalize_text(row.get('control_cabinet_no')),
                        control_cabinet_factory=ExcelImportCommon.normalize_text(row.get('control_cabinet_factory')),
                        control_cabinet_certification_no=ExcelImportCommon.normalize_text(row.get('control_cabinet_certification_no')),
                        control_cabinet_create_date=ExcelImportCommon.normalize_datetime(row.get('control_cabinet_create_date'), '控制柜制造日期'),
                        data_scope=ExcelImportCommon.normalize_text(row.get('data_scope')),
                        menu_check_strictly=ExcelImportCommon.normalize_int(row.get('menu_check_strictly'), '菜单树选择项是否关联显示', allow_negative=False),
                        dept_check_strictly=ExcelImportCommon.normalize_int(row.get('dept_check_strictly'), '部门树选择项是否关联显示', allow_negative=False),
                        status=ExcelImportCommon.normalize_status(row.get('status'), '角色状态'),
                        remark=ExcelImportCommon.normalize_text(row.get('remark')),
                        create_time=datetime.now(),
                        update_time=datetime.now(),
                    )
                    await Control_cabinet_infoDao.add_control_cabinet_info_dao(query_db, add_control_cabinet_info)
                except FieldValidationError as exc:
                    error_message = exc.message or str(exc) or exc.__class__.__name__
                    raise ServiceException(message=f'第{excel_row_no}行导入失败：{error_message}') from exc
                except ServiceException as exc:
                    error_message = exc.message or str(exc) or exc.__class__.__name__
                    raise ServiceException(message=f'第{excel_row_no}行导入失败：{error_message}') from exc
                except Exception as exc:
                    error_message = str(exc) or repr(exc) or exc.__class__.__name__
                    raise ServiceException(message=f'第{excel_row_no}行导入失败：{error_message}') from exc

            await query_db.commit()
            return CrudResponseModel(is_success=True, message='导入成功')
        except Exception as exc:
            await query_db.rollback()
            raise exc
