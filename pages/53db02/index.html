# 布尔值

  1. true 或者false两个值

  2. 使用bool()关键字来进行布尔转换

  3. # int ---> bool
i = 100
print(bool(i))  # True  # 非零即True
i1 = 0
print(bool(i1))  # False 零即False
# str ---> bool
s1 = '太白'
s2 = ''
print(bool(s1))  # True 非空即True
print(bool(s2))  # False
