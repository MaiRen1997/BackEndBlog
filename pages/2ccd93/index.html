各参数作用：

  • alias_generator=to_camel：允许进行驼峰转换
  • from_attributes=True：允许从 ORM 对象（如 SQLAlchemy）或其他类实例创建模型
  • populate_by_name=True：既能用字段名也能用别名来初始化/访问数据

# alias_generator的使用

to_camel 是 Pydantic 提供的驼峰命名转换器，将 Python 的蛇形命名（snake_case）转换为驼峰命名（camelCase）：

典型应用场景：

  • Python 后端习惯用 snake_case
  • 前端 JavaScript 习惯用 camelCase
  • 自动转换，无需手动映射
from pydantic.alias_generators import to_camel

# 实际使用效果
class UserModel(BaseModel):
    model_config = ConfigDict(alias_generator=to_camel)
    
    user_name: str      # Python中保持 snake_case
    user_age: int       # Python中保持 snake_case
    phone_number: str   # Python中保持 snake_case

# JSON 序列化/反序列化时自动转换
user = UserModel(user_name="张三", user_age=25, phone_number="13800138000")
print(user.model_dump(by_alias=True))
# 输出: {"userName": "张三", "userAge": 25, "phoneNumber": "13800138000"}

# 以上三者者配合使用的完整示例

  1. 前端可以通过驼峰传入数值alias_generator=to_camel
  2. 可以从数据库取值，直接放入到模型中from_attributes=True
  3. 可以通过小驼峰、下划线两种方式访问变量populate_by_name=True
from pydantic import BaseModel, ConfigDict
from pydantic.alias_generators import to_camel

# 1. 定义 ORM 模型（数据库）
class UserDB:
    def __init__(self, user_name, user_age, user_email):
        self.user_name = user_name
        self.user_age = user_age
        self.user_email = user_email

# 2. 定义 Pydantic 模型（API 响应）
class UserResponse(BaseModel):
    model_config = ConfigDict(
        alias_generator=to_camel,      # 前端用 camelCase
        from_attributes=True,           # 可以从 ORM 对象转换
        populate_by_name=True           # 可以同时使用字段名和别名
    )
    user_name: str
    user_age: int
    user_email: str | None = None

# 3. 实际使用场景
# 从数据库获取数据
db_user = UserDB("张三", 25, "zhangsan@example.com")

# 转换为 API 响应模型（from_attributes 起作用）
response = UserResponse.model_validate(db_user)

# 序列化为 JSON 返回给前端（alias 起作用）
json_response = response.model_dump(by_alias=True)
print(json_response)
# 输出: {"userName": "张三", "userAge": 25, "userEmail": "zhangsan@example.com"}

# 同时，内部代码可以用原始字段名访问（populate_by_name 起作用）
print(response.user_name)  # ✅ "张三"
print(response.userName)   # ✅ 也可以访问 "张三"

# NotBlank的使用

# 定义

@NotBlank 是一个自定义验证装饰器（来自 pydantic_validation_decorator 库），用于验证字段不能为空或空白字符串：

from datetime import datetime
from pydantic import BaseModel, ConfigDict, Field
from pydantic.alias_generators import to_camel
from pydantic_validation_decorator import NotBlank

class Ard_infoModel(BaseModel):
    """
    ard信息表对应pydantic模型
    """
    model_config = ConfigDict(alias_generator=to_camel, from_attributes=True, populate_by_name= True)
    contract_no: str | None = Field(default=None, description='合同号')
    update_time: datetime | None = Field(default=None, description='更新时间')
    
    @NotBlank(field_name='contract_no', message='合同编号不能为空')
    def get_contract_no(self) -> str | None:
        return self.contract_no
    @NotBlank(field_name='start_date', message='更新时间不能为空')
    def get_update_time(self) -> datetime | None:
        return self.start_date

    def validate_fields(self) -> None:
        self.get_contract_no()
        self.get_update_time()

class DynamicResponseModel(ResponseBaseModel, Generic[T]):
    """
    动态响应模型
    """

    model_config = ConfigDict(alias_generator=to_camel)

    def __class_getitem__(cls, item: Any) -> Any | Self:
        """
        当使用 DynamicResponseModel[Item] 语法时，动态创建一个包含所有字段的新模型
        """
        # 检查是否已经为该类型创建了模型
        if not hasattr(cls, '_cached_models'):
            cls._cached_models = {}

        if item in cls._cached_models:
            return cls._cached_models[item]

        # 检查item是否为Pydantic模型
        if not hasattr(item, 'model_fields'):
            raise TypeError(f'{item} 不是一个Pydantic模型，请使用Pydantic模型作为泛型参数')

        # 获取ResponseBaseModel的字段
        base_fields = {}
        for field_name, field in cls.model_fields.items():
            base_fields[field_name] = (field.annotation, field)

        # 获取泛型类型的字段
        item_fields = {}
        for field_name, field in item.model_fields.items():
            item_fields[field_name] = (field.annotation, field)

        # 合并所有字段
        all_fields = {**base_fields, **item_fields}

        # 动态创建新模型
        new_model = create_model(
            f'DynamicResponseModel[{item.__name__}]', __base__=cls, __config__=cls.model_config, **all_fields
        )

        # 缓存模型
        cls._cached_models[item] = new_model

        return new_model
