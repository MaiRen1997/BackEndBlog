  2. minio_client.stat_object用于检查文件是否存在于该桶中
    1. 通过调用 MinIO 的 stat_object 方法获取对象的元数据（如大小、修改时间、ETag 等）。
    2. 如果获取成功，说明对象存在，返回 True。
    3. 如果获取失败，捕获异常并判断失败原因是否是“对象不存在”，若是则返回 False；若是其他错误，则继续抛出异常

    FILE_LABEL_MAP = {
        'safety_gear_certification_url': '安全钳',
        'certification_url': '合格证',

    @classmethod
    def _get_minio_client_and_endpoint(cls) -> tuple[Minio, str]:
        endpoint = MinioConfig.minio_endpoint.strip() # 获取IP地址 + 端口号
        if not endpoint:
            raise ServiceException(message='MINIO_ENDPOINT未配置')

        normalized_endpoint = endpoint
        if normalized_endpoint.startswith('http://'):
            normalized_endpoint = normalized_endpoint[len('http://') :]
        elif normalized_endpoint.startswith('https://'):
            normalized_endpoint = normalized_endpoint[len('https://') :]

        if not MinioConfig.minio_access_key or not MinioConfig.minio_secret_key:
            raise ServiceException(message='MINIO访问密钥未配置')

        client = Minio(
            endpoint=normalized_endpoint,
            access_key=MinioConfig.minio_access_key,
            secret_key=MinioConfig.minio_secret_key,
            secure=MinioConfig.minio_secure,
        )

        return client, normalized_endpoint
    @classmethod
    def _build_public_read_policy(cls, bucket_name: str) -> str:
        return json.dumps(
            {
                'Version': '2012-10-17',  # 策略语言版本，固定为 2012-10-17（AWS S3 标准版本）
                'Statement': [ # 权限声明列表
                    {
                        'Effect': 'Allow', # 允许执行指定操作
                        'Principal': {'AWS': ['*']}, # 授权主体为所有人（* 表示匿名用户或任何 AWS 用户）
                        'Action': ['s3:GetObject'], # 允许的操作是下载/读取对象， PutObject、DeleteObject 是上传和下载
                        'Resource': [f'arn:aws:s3:::{bucket_name}/*'], # 资源范围为该存储桶下的所有对象（/* 表示所有文件）
                    }
                ],
            }
        )
    @classmethod
    def _object_exists(cls, minio_client: Minio, bucket_name: str, object_name: str) -> bool:
        try:
            minio_client.stat_object(bucket_name, object_name)
            return True
        except S3Error as exc:
            if exc.code in {'NoSuchKey', 'NoSuchObject'}:
                return False
            raise

    @classmethod
    def _next_backup_object_name(cls, minio_client: Minio, bucket_name: str, object_name: str) -> str:
        object_path = Path(object_name)
        stem = object_path.stem
        suffix = object_path.suffix
        parent = object_path.parent.as_posix()

        index = 1
        while True:
            candidate_name = f'{stem}({index}){suffix}'
            candidate_object = f'{parent}/{candidate_name}' if parent and parent != '.' else candidate_name
            if not cls._object_exists(minio_client, bucket_name, candidate_object):
                return candidate_object
            index += 1

    @classmethod
    def _rename_existing_object_if_needed(cls, minio_client: Minio, bucket_name: str, object_name: str) -> None:
        if not cls._object_exists(minio_client, bucket_name, object_name):
            return

        backup_object_name = cls._next_backup_object_name(minio_client, bucket_name, object_name)
        minio_client.copy_object(
            bucket_name=bucket_name,
            object_name=backup_object_name,
            source=CopySource(bucket_name=bucket_name, object_name=object_name),
        )
        minio_client.remove_object(bucket_name=bucket_name, object_name=object_name)
        
    @classmethod
    def _object_exists(cls, minio_client: Minio, bucket_name: str, object_name: str) -> bool:
        try:
            minio_client.stat_object(bucket_name, object_name)
            return True
        except S3Error as exc:
            if exc.code in {'NoSuchKey', 'NoSuchObject'}:
                return False
            raise
    @classmethod
        async def upload_safety_materials_file_services(
            cls,
            query_db: AsyncSession,
            contract_no: str,
            file_type: str,
            file: UploadFile,
        ) -> UploadResponseModel:
            """
            上传客货梯安全资料证书文件到MinIO
            """
            if not contract_no:
                raise ServiceException(message='合同号不能为空')
            if file_type not in cls.FILE_LABEL_MAP:
                raise ServiceException(message='不支持的文件类型')
            if not file.filename:
                raise ServiceException(message='文件名不能为空')

            safety_materials_info = await cls.safety_materials_detail_services(query_db, contract_no)
            if not safety_materials_info.contract_no:
                raise ServiceException(message='客货梯安全资料不存在，请先新增资料')

            bucket_name = MinioConfig.minio_bucket_name.strip()
            if not bucket_name:
                raise ServiceException(message='MINIO_BUCKET_NAME未配置')

            file_content = await file.read()
            if not file_content:
                raise ServiceException(message='上传文件不能为空')

            minio_client, endpoint = cls._get_minio_client_and_endpoint()

            file_extension = Path(file.filename).suffix
            object_dir = contract_no.strip()
            label = cls.FILE_LABEL_MAP[file_type]
            base_filename = f'{contract_no}{label}'
            object_name = f'{object_dir}/{base_filename}{file_extension}'

            try:
                if not minio_client.bucket_exists(bucket_name):
                    minio_client.make_bucket(bucket_name)

                minio_client.set_bucket_policy(bucket_name, cls._build_public_read_policy(bucket_name))
                cls._rename_existing_object_if_needed(minio_client, bucket_name, object_name)
                # 使用的是 MinIO 客户端中配置的 Access Key / Secret Key（即管理员或服务账号的凭证），
                # 这些凭证通常具有完整的读写权限（如 admin 或 readwrite 用户）
                minio_client.put_object( 
                    bucket_name=bucket_name,
                    object_name=object_name,
                    data=BytesIO(file_content),
                    length=len(file_content),
                    content_type=file.content_type or 'application/octet-stream',
                )
            except S3Error as exc:
                logger.error(f'MinIO上传失败: {exc}')
                raise ServiceException(message='文件上传失败') from exc

            protocol = 'https' if MinioConfig.minio_secure else 'http'
            file_url = f'{protocol}://{endpoint}/{bucket_name}/{object_name}'

            return UploadResponseModel(
                file_name=object_name,
                new_file_name=Path(object_name).name,
                original_filename=file.filename,
                url=file_url,
            )

@classmethod
    async def save_safety_materials_file_urls_services(
        cls,
        query_db: AsyncSession,
        page_object: SaveSafetyMaterialsFileUrlModel,
        current_user_name: str,
    ) -> CrudResponseModel:
        """
        保存客货梯安全资料文件url
        """
        safety_materials_info = await cls.safety_materials_detail_services(query_db, page_object.contract_no)
        if not safety_materials_info.contract_no:
            raise ServiceException(message='客货梯安全资料不存在')

        update_payload = {
            'contract_no': page_object.contract_no,
            'safety_gear_certification_url': page_object.safety_gear_certification_url,
            'certification_url': page_object.certification_url,
            'buffer_certification_url': page_object.buffer_certification_url,
            'speed_limit_certification_url': page_object.speed_limit_certification_url,
            'machine_certification_url': page_object.machine_certification_url,
            'update_by': current_user_name,
            'update_time': datetime.now(),
        }

        try:
            await Safety_materialsDao.edit_safety_materials_dao(query_db, update_payload)
            await query_db.commit()
            return CrudResponseModel(is_success=True, message='文件链接保存成功')
        except Exception as exc:
            await query_db.rollback()
            raise exc

            
@classmethod
    async def safety_materials_detail_services(cls, query_db: AsyncSession, contract_no: str) -> Safety_materialsModel:

        :param contract_no: 合同号

        safety_materials = await Safety_materialsDao.get_safety_materials_detail_by_id(query_db, contract_no=contract_no)
        result = Safety_materialsModel(**CamelCaseUtil.transform_result(safety_materials)) if safety_materials else Safety_materialsModel()

        return result

  {
    key: 'machineCertificationUrl',
    label: '主机合格证',
    uploader: uploadMachineCertificationFile,
  },
];

const toUploadFile = (url?: string): UploadFile[] => {
  if (!url) {
    return [];
  }

  const fileName = decodeURIComponent(url.split('/').pop() || 'uploaded-file');
  return [
    {
      uid: url,
      name: fileName,
      status: 'done',
      url,
    },
  ];
};

const getUploadResultUrl = (response: unknown): string | undefined => {
  const result = response as UploadApiResponse;
  return result.url || result.data?.url;
};

const toPayload = (contractNo: string, fileUrls: Partial<Record<FileFieldKey, string | undefined>>): MaterialFileUrlPayload => {
  return {
    contractNo,
    safetyGearCertificationUrl: fileUrls.safetyGearCertificationUrl,
    certificationUrl: fileUrls.certificationUrl,
    bufferCertificationUrl: fileUrls.bufferCertificationUrl,
    speedLimitCertificationUrl: fileUrls.speedLimitCertificationUrl,
    machineCertificationUrl: fileUrls.machineCertificationUrl,
  };
};

export interface SafetyMaterialFileUploadModalProps {
  open: boolean;
  setOpen: (open: boolean) => void;
  record?: SafetyMaterialItem;
  onSuccess?: () => void | Promise<void>;
}

const SafetyMaterialFileUploadModal: React.FC<SafetyMaterialFileUploadModalProps> = ({
  open,
  setOpen,
  record,
  onSuccess,
}) => {
  const { message } = App.useApp();
  const contractNo = record?.contractNo || '';
  const [fileUrls, setFileUrls] = useState<Partial<Record<FileFieldKey, string | undefined>>>({});
  const [fileLists, setFileLists] = useState<Partial<Record<FileFieldKey, UploadFile[]>>>({});
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    if (!open) {
      return;
    }

    const nextUrls: Partial<Record<FileFieldKey, string | undefined>> = {
      safetyGearCertificationUrl: record?.safetyGearCertificationUrl,
      certificationUrl: record?.certificationUrl,
      bufferCertificationUrl: record?.bufferCertificationUrl,
      speedLimitCertificationUrl: record?.speedLimitCertificationUrl,
      machineCertificationUrl: record?.machineCertificationUrl,
    };
    setFileUrls(nextUrls);
    setFileLists({
      safetyGearCertificationUrl: toUploadFile(nextUrls.safetyGearCertificationUrl),
      certificationUrl: toUploadFile(nextUrls.certificationUrl),
      bufferCertificationUrl: toUploadFile(nextUrls.bufferCertificationUrl),
      speedLimitCertificationUrl: toUploadFile(nextUrls.speedLimitCertificationUrl),
      machineCertificationUrl: toUploadFile(nextUrls.machineCertificationUrl),
    });
  }, [open, record]);

  const fieldPropsMap = useMemo(() => {
    const map: Partial<Record<FileFieldKey, UploadProps>> = {};

    fileFieldMeta.forEach((field) => {
      map[field.key] = {
        name: 'file',
        maxCount: 1,
        customRequest: async (options) => {
          if (!contractNo) {
            message.error('合同号不能为空，请先选择一条资料');
            options.onError?.(new Error('contract_no_empty'));
            return;
          }

          try {
            const uploadFile = options.file as File;
            const response = await field.uploader(contractNo, uploadFile);
            const url = getUploadResultUrl(response);

            if (!url) {
              throw new Error('未获取到文件地址');
            }

            setFileUrls((prev) => ({
              ...prev,
              [field.key]: url,
            }));
            setFileLists((prev) => ({
              ...prev,
              [field.key]: [
                {
                  uid: url,
                  name: decodeURIComponent(url.split('/').pop() || uploadFile.name || 'uploaded-file'),
                  status: 'done',
                  url,
                },
              ],
            }));
            options.onSuccess?.(response as object);
            message.success(`${field.label}上传成功`);
          } catch (error) {
            const err = error as Error;
            options.onError?.(err);
          }
        },
      };
    });

    return map;
  }, [contractNo, message]);

  const handleSubmit = async () => {
    if (!contractNo) {
      message.error('合同号不能为空，请先选择一条资料');
      return;
    }

    setSaving(true);
    try {
      await saveSafetyMaterialFileUrls(toPayload(contractNo, fileUrls));
      message.success('文件链接保存成功');
      setOpen(false);
      await onSuccess?.();
    } finally {
      setSaving(false);
    }
  };

  return (
    <Modal
      title="上传安全资料文件"
      open={open}
      onCancel={() => setOpen(false)}
      onOk={() => void handleSubmit()}
      okText="提交"
      cancelText="取消"
      confirmLoading={saving}
      width={760}
    >
      <Space direction="vertical" size={16} style={{ width: '100%' }}>
        <Typography.Text>
          合同号：{contractNo || '-'}
        </Typography.Text>
        {fileFieldMeta.map((field) => {
          return (
            <div key={field.key}>
              <Space style={{ width: '100%', justifyContent: 'space-between' }}>
                <Typography.Text strong>{field.label}</Typography.Text>
                {fileUrls[field.key] ? (
                  <a href={fileUrls[field.key]} target="_blank" rel="noreferrer">
                    预览
                  </a>
                ) : (
                  <Typography.Text type="secondary">未上传</Typography.Text>
                )}
              </Space>
              <Upload
                {...fieldPropsMap[field.key]}
                fileList={fileLists[field.key] || []}
                onRemove={() => {
                  setFileUrls((prev) => ({
                    ...prev,
                    [field.key]: undefined,
                  }));
                  setFileLists((prev) => ({
                    ...prev,
                    [field.key]: [],
                  }));
                  return true;
                }}
              >
                <Button icon={<UploadOutlined />}>上传{field.label}</Button>
              </Upload>
            </div>
          );
        })}
      </Space>
    </Modal>
  );
};

export default SafetyMaterialFileUploadModal;
