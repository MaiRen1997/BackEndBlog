## 封装方法

1. 默认是None，同步全量数据
2. 如果传值，则只通过相应几条数据

```python
from config.database import AsyncSessionLocal
from module_certification.service.cbusSync_sqlserver_sync_service import CbusSyncSqlServerSyncService


async def sync_cbus_from_sqlserver(limit = None ) -> None:
    """
    同步SQL Server的CBUS数据到本地表（可选）
    """
    await _sync_cbus_from_sqlserver(limit=limit)

    
async def _sync_cbus_from_sqlserver(limit=None) -> None:
    logger.info(
        f'🚚 开始执行CBUS同步任务，SQL Server={SqlServerConfig.sqlserver_host}:{SqlServerConfig.sqlserver_port}, '
        f'limit={limit}'
    )
    if SqlServerConfig.sqlserver_host in {'127.0.0.1', 'localhost'}:
        logger.warning('⚠️ SQL Server地址仍是本机地址，请检查 Docker 环境中的 SQLSERVER_* 配置')

    async with AsyncSessionLocal() as session:
        await CbusSyncSqlServerSyncService.sync_to_cbus_table(session, limit=limit)
```
