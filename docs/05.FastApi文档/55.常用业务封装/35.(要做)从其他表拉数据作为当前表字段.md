## service 实现

```python
@classmethod
    async def sync_contractManage_services(
        cls,
        query_db: AsyncSession,
        contract_manage_id: int,
        sync_object: SyncContractmanageModel,
    ) -> CrudResponseModel:
        contract_manage = await cls.contractManage_detail_services(query_db, contract_manage_id)
        if not contract_manage.id:
            raise ServiceException(message='合格证管理不存在')

        if not contract_manage.contract_no:
            raise ServiceException(message='合同编号不能为空，无法执行同步')

        payload_map = await cls._sync_payload_by_contract_no(query_db, contract_manage.contract_no)
        exist_file_count = await cls._enrich_payload_and_count_files(query_db, payload_map)

        update_payload = {
            'id': contract_manage.id,
            'exist_file': exist_file_count,
            'update_time': datetime.now(),
            **payload_map,
        }
        if sync_object.required_date is not None:
            update_payload['required_date'] = sync_object.required_date

        try:
            await ContractmanageDao.edit_contractManage_dao(query_db, update_payload)
            await query_db.commit()
        except Exception as exc:
            await query_db.rollback()
            raise exc

        return CrudResponseModel(is_success=True, message='同步成功')
```

## _sync_payload_by_contract_no方法实现

```python
class ContractmanageService:
    """
    合格证管理模块服务层
    """

    SAFETY_MATERIALS_URL_KEYS = {
        'safety_gear_certification_url',
        'certification_url',
        'buffer_certification_url',
        'speed_limit_certification_url',
        'machine_certification_url',
    }

    RELATED_TABLE_MAPPING: dict[str, tuple[type, str]] = {
        'safety_materials': (SafetyMaterials, 'contract_no'),
        'cbus_params': (CbusSyncTable, 'contract_no'),
        'control_cabinet_info': (ControlCabinetInfo, 'contract_no'),
        'contract_door_info': (ContractDoorManage, 'contract_no'),
        'smart_board_info': (SmartBoardInfo, 'contract_no'),
        'lvct_board_info': (LvctBoardInfo, 'contract_no'),
        'ard_production_info': (ArdProductionInfo, 'contract_no'),
        'ic_card_info': (IcCardInfo, 'contract_no'),
        'wire_rope_info': (WireRopeInfo, 'contract_no'),
    }
    
    
@classmethod
    async def _sync_payload_by_contract_no(cls, query_db: AsyncSession, contract_no: str) -> dict[str, dict[str, Any]]:
        payload_map: dict[str, dict[str, Any]] = {}

        for target_field, (model_cls, contract_field) in cls.RELATED_TABLE_MAPPING.items():
            db_row = await cls._get_row_by_contract_no(query_db, model_cls, contract_no, contract_field)
            payload_map[target_field] = cls._model_to_dict(db_row) if db_row else {}

        return payload_map
```

### 通过map循环，获取数据

```python
@classmethod
    async def _get_row_by_contract_no(
        cls,
        query_db: AsyncSession,
        model_cls: type,
        contract_no: str,
        contract_field: str = 'contract_no',
    ) -> Any | None:
        column = getattr(model_cls, contract_field)
        result = await query_db.execute(select(model_cls).where(column == contract_no))
        return result.scalars().first()
```
