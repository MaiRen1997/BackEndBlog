## service实现

### 规范下载地址

```python
@classmethod
    def _normalize_download_url(cls, url: str) -> str:
        """
        规范化下载地址，避免中文和空格等路径字符导致urllib请求失败。
        """
        stripped = str(url or '').strip()
        if not stripped:
            return stripped

        parsed = urlsplit(stripped)
        encoded_path = quote(parsed.path, safe='/%:@')
        encoded_query = quote(parsed.query, safe='=&%:/?+-._~')
        return urlunsplit((parsed.scheme, parsed.netloc, encoded_path, encoded_query, parsed.fragment))
```

### 根据url下载文件

```python
 @classmethod
    async def _download_pdf_bytes(cls, url: str) -> bytes:
        normalized_url = cls._normalize_download_url(url)
        request = Request(
            normalized_url,
            headers={
                'User-Agent': 'Mozilla/5.0 (compatible; ContractmanageService/1.0)',
                'Accept': 'application/pdf,application/octet-stream,*/*;q=0.8',
            },
        )
        try:
            with urlopen(request, timeout=30) as response:
                content = response.read()
                if not content:
                    raise ServiceException(message=f'证书地址内容为空: {normalized_url}')
                return content
        except ssl.SSLError as ssl_exc:
            if normalized_url.lower().startswith('https://'):
                # 内网常见自签名证书场景，SSL校验失败时回退为不校验证书重试一次。
                try:
                    insecure_context = ssl._create_unverified_context()
                    with urlopen(request, timeout=30, context=insecure_context) as response:
                        content = response.read()
                        if not content:
                            raise ServiceException(message=f'证书地址内容为空: {normalized_url}')
                        return content
                except Exception as retry_exc:
                    logger.error(f'证书下载失败(SSL重试后仍失败): url={normalized_url}, error={retry_exc}')
                    raise ServiceException(message=f'证书文件下载失败: {normalized_url}') from retry_exc
            logger.error(f'证书下载SSL失败: url={normalized_url}, error={ssl_exc}')
            raise ServiceException(message=f'证书文件下载失败: {normalized_url}') from ssl_exc
        except Exception as exc:
            logger.error(f'证书下载失败: url={normalized_url}, error={exc}')
            raise ServiceException(message=f'证书文件下载失败: {normalized_url}') from exc
```

### 将下载的文件内容进行合并

```python
@classmethod
    def _merge_pdf_bytes(cls, pdf_contents: list[bytes]) -> bytes:
        writer = PdfWriter()
        for pdf_content in pdf_contents:
            try:
                reader = PdfReader(BytesIO(pdf_content))
                for page in reader.pages:
                    writer.add_page(page)
            except Exception as exc:
                raise ServiceException(message='存在不可解析的PDF文件，无法合并打印') from exc

        if not writer.pages:
            raise ServiceException(message='未获取到可合并的PDF页面')

        output = BytesIO()
        writer.write(output)
        return output.getvalue()
```

### 获取url,并下载和合并

```python
@classmethod
    async def print_contractManage_services(cls, query_db: AsyncSession, contract_manage_id: int) -> CrudResponseModel:
        contract_manage_do = await ContractmanageDao.get_contractManage_detail_by_id(query_db, contract_manage_id)
        if not contract_manage_do:
            raise ServiceException(message='合格证管理不存在')

        contract_manage = ContractmanageModel(**CamelCaseUtil.transform_result(contract_manage_do))
        if not contract_manage.exist_file:
            raise ServiceException(message='当前记录暂无可打印证书')

        all_urls: list[str] = []
        for field_name in [
            'safety_materials',
            'cbus_params',
            'control_cabinet_info',
            'contract_door_info',
            'smart_board_info',
            'lvct_board_info',
            'ard_production_info',
            'ic_card_info',
            'wire_rope_info',
        ]:
            payload = getattr(contract_manage, field_name, None)
            if isinstance(payload, dict):
                all_urls.extend(cls._collect_certification_urls_from_json_payload(payload))

        unique_urls: list[str] = []
        seen: set[str] = set()
        for url in all_urls:
            if url in seen:
                continue
            seen.add(url)
            unique_urls.append(url)

        if not unique_urls:
            raise ServiceException(message='未在同步数据中找到可打印的证书链接')

        pdf_contents = [await cls._download_pdf_bytes(url) for url in unique_urls]
        merged_pdf = cls._merge_pdf_bytes(pdf_contents)
        contract_no = contract_manage.contract_no or f'contract_{contract_manage_id}'
        merged_file_url = await cls._upload_merged_pdf_to_minio(contract_no, merged_pdf)

        update_payload = {
            'id': contract_manage.id,
            'file_url': merged_file_url,
            'update_time': datetime.now(),
        }
        try:
            await ContractmanageDao.edit_contractManage_dao(query_db, update_payload)
            await query_db.commit()
        except Exception as exc:
            await query_db.rollback()
            raise exc

        return CrudResponseModel(is_success=True, message='打印文件生成成功')
```

### 将合并的文件，上传到minio

1. {now.strftime("%Y/%m/%d")}，会生成一个`/年份/月份/日`的路径，用于区分年月日的

```python
@classmethod
    async def _upload_merged_pdf_to_minio(cls, contract_no: str, file_bytes: bytes) -> str:
        bucket_name = MinioConfig.minio_bucket_name.strip()
        if not bucket_name:
            raise ServiceException(message='MINIO_BUCKET_NAME未配置')

        minio_client, endpoint = cls._get_minio_client_and_endpoint()
        now = datetime.now()
        object_name = f'certification/{now.strftime("%Y/%m/%d")}/{contract_no}_{uuid4().hex}.pdf'

        try:
            if not minio_client.bucket_exists(bucket_name):
                minio_client.make_bucket(bucket_name)

            minio_client.set_bucket_policy(bucket_name, cls._build_public_read_policy(bucket_name))
            minio_client.put_object(
                bucket_name=bucket_name,
                object_name=object_name,
                data=BytesIO(file_bytes),
                length=len(file_bytes),
                content_type='application/pdf',
            )
        except S3Error as exc:
            logger.error(f'MinIO上传失败: {exc}')
            raise ServiceException(message='合并证书文件上传失败') from exc

        protocol = 'https' if MinioConfig.minio_secure else 'http'
        return f'{protocol}://{endpoint}/{bucket_name}/{object_name}'
```
