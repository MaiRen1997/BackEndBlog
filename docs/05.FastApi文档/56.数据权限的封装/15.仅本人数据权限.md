## 在controller中设置

### 改动点

1. `从sqlalchemy中引入ColumnElement`

2. `from common.aspect.data_scope import DataScopeDependency`

3. `from module_certification.entity.do.certification_do import CertificateManange`

4. 在方法定义中，增加形参`data_scope_sql: Annotated[ColumnElement, DataScopeDependency(CertificateManange)],`

5. 将形参传递到`get_certification_list_services`

6. ```PYTHON
   @certification_controller.get(
       '/list',
       summary='获取证书记录分页列表接口',
       description='用于获取证书记录分页列表',
       response_model=PageResponseModel[CertificationModel],
       dependencies=[UserInterfaceAuthDependency('certification:certification:list')],
   )
   async def get_certification_certification_list(
       request: Request,
       certification_page_query: Annotated[CertificationPageQueryModel, Query()],
       query_db: Annotated[AsyncSession, DBSessionDependency()],
       data_scope_sql: Annotated[ColumnElement, DataScopeDependency(CertificateManange)],
   ) -> Response:
       # 获取分页数据
       certification_page_query_result = await CertificationService.get_certification_list_services(query_db, certification_page_query, data_scope_sql, is_page=True)
       logger.info('获取成功')
   
       return ResponseUtil.success(model_content=certification_page_query_result)
   
   ```

### data_scope_sql的来源

`data_scope_sql` 是 **FastAPI 执行 `DataScopeDependency(CertificateManange)` 后自动注入的返回值**。

后端根据当前登录用户的角色、部门、用户 ID，动态生成这个 SQLAlchemy 查询条件。

2. 在`Service的类方法的形参中`，`增加data_scope_sql参数`，类型是ColumnElement

3. 在方法内部的Dao调用过程中，根据形参，传入对应的实参到Dao中

4. 在删除和查询详情接口，要提前使用`get_certification_detail_by_id`预判是否有权限删除和查看

   1. ```python
      certification = await CertificationDao.get_certification_detail_by_id(query_db, id=id, data_scope_sql=data_scope_sql)

          raise ServiceException(message='证书记录不存在或无权访问')
      result = CertificationModel(**CamelCaseUtil.transform_result(certification))
          return result
      ```

   2. 由于调用的，里面包含了异常抛出，如果出现异常，会阻断操作，所以此处不需要额外操作

      1. ```python
         for id in id_list:

5. ```python
   from typing import Any
   
   from sqlalchemy import ColumnElement
   from sqlalchemy.ext.asyncio import AsyncSession
   
   from common.constant import CommonConstant
   from common.vo import CrudResponseModel, PageModel
   from exceptions.exception import ServiceException
   from module_certification.service.submitter import set_submitter
   from module_certification.dao.certification_dao import CertificationDao
   from module_certification.entity.vo.certification_vo import DeleteCertificationModel, CertificationModel, CertificationPageQueryModel
   from utils.common_util import CamelCaseUtil
   from utils.excel_util import ExcelUtil
   
   
   class CertificationService:
       """
       证书记录模块服务层
       """
   
       @classmethod
       async def get_certification_list_services(
           cls, query_db: AsyncSession, query_object: CertificationPageQueryModel, data_scope_sql: ColumnElement, is_page: bool = False
       ) -> PageModel | list[dict[str, Any]]:
           """
           获取证书记录列表信息service
   
           :param query_db: orm对象
           :param query_object: 查询参数对象
           :param is_page: 是否开启分页
           :return: 证书记录列表信息对象
           """
           certification_list_result = await CertificationDao.get_certification_list(query_db, query_object, data_scope_sql, is_page)
   
           return certification_list_result
   
   
       @classmethod
       async def add_certification_services(cls, query_db: AsyncSession, page_object: CertificationModel) -> CrudResponseModel:
           """
           新增证书记录信息service
   
           :param query_db: orm对象
           :param page_object: 新增证书记录对象
           :return: 新增证书记录校验结果
           """
           set_submitter(page_object)
           try:
               await CertificationDao.add_certification_dao(query_db, page_object)
               await query_db.commit()
               return CrudResponseModel(is_success=True, message='新增成功')
           except Exception as e:
               await query_db.rollback()
               raise e
   
       @classmethod
       async def edit_certification_services(cls, query_db: AsyncSession, page_object: CertificationModel, data_scope_sql: ColumnElement) -> CrudResponseModel:
           """
           编辑证书记录信息service
   
           :param query_db: orm对象
           :param page_object: 编辑证书记录对象
           :return: 编辑证书记录校验结果
           """
           edit_certification = page_object.model_dump(exclude_unset=True, exclude={'user_id', 'user_name', 'create_by', 'create_user_name'})
           certification_info = await cls.certification_detail_services(query_db, page_object.id, data_scope_sql)
           if certification_info.id:
               try:
                   await CertificationDao.edit_certification_dao(query_db, edit_certification)
                   await query_db.commit()
                   return CrudResponseModel(is_success=True, message='更新成功')
               except Exception as e:
                   await query_db.rollback()
                   raise e
           else:
               raise ServiceException(message='证书记录不存在')
   
       @classmethod
       async def delete_certification_services(cls, query_db: AsyncSession, page_object: DeleteCertificationModel, data_scope_sql: ColumnElement) -> CrudResponseModel:
           """
           删除证书记录信息service
   
           :param query_db: orm对象
           :param page_object: 删除证书记录对象
           :return: 删除证书记录校验结果
           """
           if page_object.ids:
               id_list = page_object.ids.split(',')
               try:
                   for id in id_list:
                       await cls.certification_detail_services(query_db, int(id), data_scope_sql)
                   for id in id_list:
                       await CertificationDao.delete_certification_dao(query_db, CertificationModel(id=id))
                   await query_db.commit()
                   return CrudResponseModel(is_success=True, message='删除成功')
               except Exception as e:
                   await query_db.rollback()
                   raise e
           else:
               raise ServiceException(message='传入为空')
   
       @classmethod
       async def certification_detail_services(cls, query_db: AsyncSession, id: int, data_scope_sql: ColumnElement) -> CertificationModel:
           """
           获取证书记录详细信息service
   
           :param query_db: orm对象
           :param id:
           :return: 对应的信息
           """
           certification = await CertificationDao.get_certification_detail_by_id(query_db, id=id, data_scope_sql=data_scope_sql)
           if certification is None:
               raise ServiceException(message='证书记录不存在或无权访问')
           result = CertificationModel(**CamelCaseUtil.transform_result(certification))
   
           return result
   
       @staticmethod
       async def export_certification_list_services(certification_list: list) -> bytes:
           """
           导出证书记录信息service
   
           :param certification_list: 证书记录信息列表
           :return: 证书记录信息对应excel的二进制数据
           """
           # 创建一个映射字典，将英文键映射到中文键
           mapping_dict = {
               'id': '',
               'certificationNo': '证书编号',
               'reportNo': '报告编号',
               'productName': '产品名称',
               'productSerial': '产品型号',
               'factory': '制造单位',
               'issuingAuthority': '发证机构',
               'fileUrl': '证书文件链接',
               'startDate': '证件生效日期',
               'expiredTime': '证件过期日期',
               'status': '状态',
               'deptId': '部门id',
           }
           binary_data = ExcelUtil.export_list2excel(certification_list, mapping_dict)
   
           return binary_data
   ```

## 在dao中的设置

### 代码增加部分

1. 从sqlalchemy中引入ColumnElement

2. 在类方法中，增加data_scope_sql参数，类型是ColumnElement

3. 在sql方法中的where查询语句中，增加data_scope_sql

4. ```python
   from sqlalchemy import ColumnElement, delete, select, update
   class CertificationDao:
       """
       证书记录模块数据库操作层
       """
   
       @classmethod
       async def get_certification_detail_by_id(cls, db: AsyncSession, id: int, data_scope_sql: ColumnElement) -> CertificateManange | None:
           """
           根据获取证书记录详细信息
   
           :param db: orm对象
           :param id:
           :return: 证书记录信息对象
           """
           certification_info = (
               (
                   await db.execute(
                       select(CertificateManange)
                       .where(
                           CertificateManange.id == id,
                           data_scope_sql,
                       )
                   )
               )
               .scalars()
               .first()
           )
   
           return certification_info
   ```
