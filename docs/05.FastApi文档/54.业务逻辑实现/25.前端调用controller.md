## 完整代码

```js
import request from '@/utils/request'

// 查询用户会话关联列表
export function listSessions(query) {
  return request({
    url: '/sessions/sessions/list',
    method: 'get',
    params: query
  })
}

// 查询用户会话关联详细
export function getSessions(id) {
  return request({
    url: '/sessions/sessions/' + id,
    method: 'get'
  })
}

// 新增用户会话关联
export function addSessions(data) {
  return request({
    url: '/sessions/sessions',
    method: 'post',
    data: data
  })
}

// 修改用户会话关联
export function updateSessions(data) {
  return request({
    url: '/sessions/sessions',
    method: 'put',
    data: data
  })
}

// 删除用户会话关联
export function delSessions(id) {
  return request({
    url: '/sessions/sessions/' + id,
    method: 'delete'
  })
}
```
